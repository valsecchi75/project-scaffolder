# rm ricorsivo+force in ogni ordine di flag (cluster combinato, es. -rf/-fr)
rm[[:space:]]+-[a-zA-Z]*r[a-zA-Z]*f
rm[[:space:]]+-[a-zA-Z]*f[a-zA-Z]*r
# rm ricorsivo+force con flag separati (es. -r -f / -f -r), in ogni ordine
rm[[:space:]]+.*-r\b.*-f\b
rm[[:space:]]+.*-f\b.*-r\b
# rm ricorsivo+force in forma estesa (--recursive --force), in ogni ordine
rm[[:space:]]+.*--recursive\b.*--force\b
rm[[:space:]]+.*--force\b.*--recursive\b
# PowerShell recurse+force delete
Remove-Item[[:space:]].*-Recurse
